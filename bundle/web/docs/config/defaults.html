# =============================================================================
# Code Grading Suite: default configuration (suite v5.0.0)
# =============================================================================
#
# Every adjustable part of scoring lives in this file: weights, the questions
# that adjust them, presets, gates, criticality zones, targets, maturity
# levels and how actions are prioritized.
#
# HOW TO CUSTOMIZE: you rarely need to edit this file. Configs stack, each
# layer overriding the one before:
#
#     config/defaults.toml   ← this file (suite defaults)
#     your-org.toml          ← optional: house rules shared across projects
#     project.toml           ← per project (projects/<name>/project.toml in your grading
#                               repo): answers, zones, overrides
#
#     python3 tools/gradekit.py resolve --config your-org.toml --config project.toml
#
# Tables merge key by key; lists are replaced whole. Start from
# config/project.example.toml. See TUNING.md for recipes.
#
# Org and project configs are owned by people and live in your grading repo.
# Agents read them but never change them.
# =============================================================================

# The version of the scoring rules, which is not always the kit's version. It changes only
# when the scoring rules change, so releases that only add tools, fields or wording (such as
# 4.2.2 and 4.3.0) keep config fingerprints, and older reports, valid. 5.0.0 changed what
# severity means (see ZONES); reports graded under 4.x keep their rules (config/scoring/4.2.0.toml).
suite_version = "5.0.0"

# -----------------------------------------------------------------------------
# GATES
# Gates are non-negotiable: a weighted average can't make up for them.
#   - A gate at or below `hard_floor` caps the overall score at `overall_cap`.
#   - Each maturity level sets a `gate_min` that every gate must meet.
# Gates = `always` + gates added by your answers + `extra` - `exclude`.
# -----------------------------------------------------------------------------
[gates]
always      = ["A1", "A2"]
extra       = []            # e.g. ["C4"] to make tests a gate for every project
exclude     = []            # removing A1 or A2 is allowed, but resolve warns loudly
hard_floor  = 3
overall_cap = 4.0


# -----------------------------------------------------------------------------
# CRITICALITY ZONES
# Not all code matters equally. Projects map paths to zones in project.toml
# (as a [[zones]] list; the settings below are in [zone_settings]):
#   [[zones]]
#   name = "payments"
#   paths = ["src/payments/**", "src/billing/**"]
#   criticality = "hot"            # hot | normal | cold
#   why = "Moves money"
# A file takes the FIRST zone whose pattern matches, so list specific zones
# first. Unmatched files are `default`. A finding takes the hottest zone among
# the files it cites.
#
# Zones say where a problem costs more, and so where the bar is higher. They don't
# change how bad a problem is: the grader judges a finding's severity by its
# consequence where the code runs (a wrong total in payments is critical; a style
# nit there is still minor). Effects:
#   - Maturity levels hold hot zones to a higher standard (see MATURITY: L2 allows
#     no critical findings there, L3 no major ones).
#   - priority_multiplier: hot-zone work is ranked first (see PRIORITIES).
#   - The grader reads hot zones in full and weighs them most when scoring.
# (Until scoring 5.0, a zone shifted severity instead: hot one level up, cold one
# down. That made one major-or-minor call move a score by 2-3 points.)
# -----------------------------------------------------------------------------
[zone_settings]
default         = "normal"
priority_multiplier = { hot = 2.0, normal = 1.0, cold = 0.5 }


# -----------------------------------------------------------------------------
# TARGETS: where improving agents stop
#   proportional: by weight used (see by_weight)
#   maximal:      10 for every applicable benchmark
#   level:L3:     the minimum each benchmark needs for that maturity level
# Per-benchmark overrides go in project.toml: [targets.overrides] F4 = 10
# -----------------------------------------------------------------------------
[targets]
mode        = "proportional"
by_weight   = { "3" = 10, "2" = 9, "1" = 8 }
limited_cap = 8             # a benchmark of limited relevance never targets above this
overrides   = {}


# -----------------------------------------------------------------------------
# MATURITY LEVELS (checked in order; a level counts only if all lower ones do)
#   overall_min             weighted overall score
#   gate_min                every gate benchmark
#   any_min                 every applicable benchmark
#   max_critical_gaps / max_major_must_have_gaps   open gaps in the gap register
#   findings_limits         open findings by severity, optionally limited to one zone
#   tens_require_enforcement   every 10 must cite enforcement evidence
#   holdout_pass_rate_min   only checked when hidden checks were run
# Any integrity flag caps maturity at `integrity_cap`.
# -----------------------------------------------------------------------------
[maturity]
integrity_cap = "L1"

[[maturity.levels]]
level = "L0"
name  = "Not viable"

[[maturity.levels]]
level       = "L1"
name        = "Prototype"
overall_min = 5.0
gate_min    = 4

[[maturity.levels]]
level             = "L2"
name              = "Usable / Beta"
overall_min       = 6.5
gate_min          = 6
any_min           = 4
max_critical_gaps = 0
findings_limits   = [ { severity = "critical", zone = "hot", max = 0 } ]

[[maturity.levels]]

gate_min                 = 8
any_min                  = 6

[[maturity.levels]]
level                    = "L4"
name                     = "Exemplary"
overall_min              = 9.3
gate_min                 = 9
any_min                  = 8
max_critical_gaps        = 0
max_major_must_have_gaps = 0
findings_limits          = [ { severity = "critical", max = 0 }, { severity = "major", max = 0 } ]
tens_require_enforcement = true
holdout_pass_rate_min    = 0.95


# -----------------------------------------------------------------------------
# PRIORITIES: how gradekit ranks improvement actions
#   score = weight × points gained (capped at target) × zone multiplier ÷ effort cost
# Actions are grouped into tiers before sorting by score:
#   tier 0: fixes a failed gate (at or below hard_floor) or a critical finding in a hot zone
#   tier 1: unblocks the next maturity level
#   tier 2: everything else below target
#   tier 3: optional (benchmark already at target)
# -----------------------------------------------------------------------------
[priorities]
effort_cost = { S = 1, M = 2, L = 4 }


# -----------------------------------------------------------------------------
# REVIEW: who may answer decision questions
# Decision questions (e.g. "drop the deprecated endpoint?") are answered in
# decisions.json next to project.toml. By default only people may approve them: an
# answer whose decided_by starts with "agent" counts as still pending.
# Set allow_agent_decisions = true to let agent reviewers approve them too.
# -----------------------------------------------------------------------------
[review]
allow_agent_decisions = false


# -----------------------------------------------------------------------------
# PUBLISHING: keeping security details out of public grading repos
# A grading repo may be public. Findings the grader marks "sensitive": true
# (typically security issues) are withheld by `gradekit redact --in-place`:
# the public report keeps their ID, benchmark, severity, status and the files
# they concern, so every score, gate and priority comes out the same, but not
# their title, quotes, commands or the wording of the actions that fix them.
# The full report is saved beside it as report.private.json, which grading
# repos never commit.
#   public               false if only trusted people can read the grading repo
#   sensitive_benchmarks validate warns about open findings here that aren't
#   warn_at              marked sensitive, at this severity or worse
# These settings never change scores, so they don't change the config hash.
# -----------------------------------------------------------------------------
[publishing]
public = true
sensitive_benchmarks = ["A2"]
warn_at = "major"


# -----------------------------------------------------------------------------
# SCHEDULE: grading on a schedule
# The grading CI's scheduled job runs `gradekit due --check-code --write-orders`,
# which writes a grade work order (projects/NAME/orders/) for each project that's
# due and sends them as a pull request. Whoever runs graders picks them up with
# `gradekit order list`.
#   grade_every_days      0: no schedule. Otherwise, due this many days after
#                         the last full grade (or now, if never graded).
#   only_if_code_changed  and only if the code's default branch has moved since
#                         the commit last graded.
# These settings never change scores, so they don't change the config hash.
# -----------------------------------------------------------------------------
[schedule]
grade_every_days = 0
only_if_code_changed = true
